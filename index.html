    </div>

        目標サイズ: <input type="number" id="targetSize" value="10"> MB

    <button id="runBtn">圧縮を開始する</button>

<script type="module">
// FFmpeg.wasmの読み込み（最新のESM版）
import { FFmpeg } from 'https://unpkg.com/@ffmpeg/ffmpeg@0.12.10/dist/esm/index.js';
import { toBlobURL } from 'https://unpkg.com/@ffmpeg/util@0.12.1/dist/esm/index.js';

const ffmpeg = new FFmpeg();
const runBtn = document.getElementById('runBtn');
const fileInput = document.getElementById('fileInput');

fileInput.onchange = (e) => {
    document.getElementById('fileName').innerText = e.target.files[0].name;
};

runBtn.onclick = async () => {
    const file = fileInput.files[0];
    const targetMB = parseFloat(document.getElementById('targetSize').value);
    if (!file) return alert("動画を選んでね");

    runBtn.disabled = true;
    document.getElementById('progress').style.display = 'block';
    
    // FFmpeg本体のロード（初回のみ時間がかかる）
    if (!ffmpeg.loaded) {
        document.getElementById('statusText').innerText = "ライブラリ読み込み中...";
        const baseURL = 'https://unpkg.com/@ffmpeg/core@0.12.6/dist/esm';
        await ffmpeg.load({
            coreURL: await toBlobURL(`${baseURL}/ffmpeg-core.js`, 'text/javascript'),
            wasmURL: await toBlobURL(`${baseURL}/ffmpeg-core.wasm`, 'application/wasm'),
        });
    }

    // ファイル書き込み
    const { name } = file;
    await ffmpeg.writeFile('input.mp4', await file.arrayBuffer());

    // 進捗の監視
    ffmpeg.on('progress', ({ progress }) => {
        const p = Math.round(progress * 100);
        document.getElementById('progressBar').style.width = p + '%';
        document.getElementById('statusText').innerText = `圧縮中: ${p}%`;
    });

    // 圧縮実行 (libx264)
    // -fs でファイルサイズを制限（MB単位をバイトに変換）
    const targetBytes = Math.floor(targetMB * 1024 * 1024);
    await ffmpeg.exec(['-i', 'input.mp4', '-b:v', '1M', '-fs', `${targetBytes}`, 'output.mp4']);

    // 結果の出力
    const data = await ffmpeg.readFile('output.mp4');
    const url = URL.createObjectURL(new Blob([data.buffer], { type: 'video/mp4' }));
    
    document.getElementById('result').innerHTML = `
        <div style="margin-top:20px; border-top:1px solid #eee; padding-top:10px;">
            <p>✅ 完了しました！</p>
            <a href="${url}" download="compressed_${name}">💾 ここを長押し/クリックで保存</a>
        </div>
    `;
    runBtn.disabled = false;
};
</script>
